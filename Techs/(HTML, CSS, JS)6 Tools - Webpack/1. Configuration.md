# Configuration

Webpack is configured in the `webpack.config.js` file, at the root of the project.

*Example:*

```javascript
const path = require('path');

module.exports = {
    mode: 'development',
    entry: './src/index.js', //Module to start the dependencies list
    output: {
        filename: 'script.js', //.js generated file name
        path: path.resolve(__dirname, 'dist'), //Generated files path
        clean: true //Clean the folder before each build
    }
};
```

## Full example

A complete configuration, with CSS, Sass, images, copied files, minimizers and dev server.

```json
{
  "scripts": {
    "dev": "webpack-dev-server",
    "prod": "cross-env NODE_ENV=production webpack"
  },
  "devDependencies": {
    "copy-webpack-plugin": "^12.0.2",
    "cross-env": "^7.0.3",
    "css-loader": "^7.1.2",
    "css-minimizer-webpack-plugin": "^7.0.0",
    "mini-css-extract-plugin": "^2.9.0",
    "sass": "^1.77.0",
    "sass-loader": "^14.2.1",
    "terser-webpack-plugin": "^5.3.10",
    "webpack": "^5.91.0",
    "webpack-cli": "^5.1.4",
    "webpack-dev-server": "^5.0.4"
  }
}
```

```javascript
const path = require('path');
const mode = process.env.NODE_ENV !== 'production';
const miniCssExtractPlugin = require('mini-css-extract-plugin');
const copyPlugin = require('copy-webpack-plugin');
const terserPlugin = require('terser-webpack-plugin');
const cssMinimizerPlugin = require('css-minimizer-webpack-plugin');

module.exports = {
    mode: mode ? 'development' : 'production',
    entry: './src/index.js',
    output: {
        filename: 'script.js',
        path: path.resolve(__dirname, 'dist')
    },
    plugins: [
        new miniCssExtractPlugin({
            filename: 'style.css'
        }),
        new copyPlugin({
            patterns: [
                { context: 'src', from: '**/*.html', noErrorOnMissing: true },
                { context: 'src', from: './assets', to: './assets', noErrorOnMissing: true }
            ]
        })
    ],
    module: {
        rules: [
            {
                test: /\.css$/,
                use: [miniCssExtractPlugin.loader, 'css-loader']
            },
            {
                test: /\.s[ac]ss$/,
                use: [miniCssExtractPlugin.loader, 'css-loader', 'sass-loader']
            },
            {
                test: /\.(png|svg|jpg|gif)$/,
                type: 'asset/resource'
            }
        ]
    },
    optimization: {
        minimizer: [
            new terserPlugin(),
            new cssMinimizerPlugin()
        ]
    },
    devServer: {
        static: './dist',
        port: 8080
    }
}
```

## Mode by environment

`NODE_ENV` is defined in the script. `cross-env` makes it work on Windows too.

*Example:*

```json
{
  "scripts": {
    "dev": "webpack-dev-server",
    "prod": "cross-env NODE_ENV=production webpack"
  }
}
```

```javascript
const isDev = process.env.NODE_ENV !== 'production';

module.exports = {
    mode: isDev ? 'development' : 'production',
};
```

## Optimization

Minimizers reduce the final files. In production, the JS is minimized by default, but setting `minimizer` replaces the default list.

*Example:*

```javascript
const terserPlugin = require('terser-webpack-plugin');
const cssMinimizerPlugin = require('css-minimizer-webpack-plugin');

module.exports = {
    optimization: {
        minimizer: [
            new terserPlugin(), //.js uglifier(minimizer)
            new cssMinimizerPlugin() //.css uglifier(minimizer)
        ]
    }
};
```

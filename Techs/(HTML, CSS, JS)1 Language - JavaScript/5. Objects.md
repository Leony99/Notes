# Objects

## Objects initializers

**Literal object notation:**

```javascript
const car = {
    brand: "BMW",
    model: "X6",
    year: 2020,
    price: 720000,
    owner: {
        name: "Leony",
        age: 23,
        code: 19232132908
    }
}
const car2 = {};
const car3 = new Object;
```

**Factory functions:**

```javascript
function createProduct(name, price, discount = 10) {
    return {
        name: name,
        price: price,
        discountPrice: price - price * (discount/100)
    }
}

const p1 = createProduct("boné", 150);
```

**Constructor functions:**

```javascript
function person(name) {
    this.name = name;

    this.talk = function() {
        console.log(`Meu nome é ${this.name}`);
    }
}

const p2 = new person("Leo");
```

**Class:**

```javascript
class Person {
    constructor(name) {
        this.name = name;
    }

    talk() {
        console.log(`Meu nome é ${this.name}`);
    }
}

const p3 = new Person("Leo");
```

## Object manipulation

```javascript
const product = new Object;

//Create/update attribute (dot/bracket)
product.name = "Chair";
product.value = 220;
product["brand"] = "Generic";

//Delete attribute (dot/bracket)
delete product.value;
delete product["brand"];
```

## Important methods

```javascript
//Return an array with the keys/values of the object
Object.entries(product);
//Return an array with the keys of the object
Object.keys(product);
//Return an array with the values of the object
Object.values(product);

//Freeze object (Cannot edit the object)
Object.freeze(product);
```

## Heritage

```javascript
class Animal {
    constructor(name) {
        this.name = name;
    }

    speak() {
        console.log(`${this.name} made some noise!`);
    }
}

class Dog extends Animal {
    constructor(name) {
        super(name);
    }

    speak() {
        console.log(`${this.name} barked!`)
    }
}

let a = new Animal("Rex");
a.speak();
let b = new Dog("Ronnie");
b.speak();
```

**Prototype:**

```javascript
Object.prototype.attr0 = "0Pro";
const avo = {attr1: "1A"};
const pai = {attr2: "2P", __proto__: avo};
const filho = {attr2: "2F", attr3: "3F", __proto__: pai};

console.log(filho.attr3, filho.attr2, filho.attr1, filho.attr0);

//Create a new object with __proto__ = parameter
const neto = Object.create(filho);
//Checks if the attribute is own or comes from a prototype
neto.hasOwnProperty('attr2');
```

## Object destructuring

```javascript
const person = {
    name: "Ana",
    age: 25,
    address: {
        city: "Rio de Janeiro - RJ",
        district: "Taquara",
        street: "Estrada do cafunda, 3211",
        house: 1
    }
}

//Extract name and age from object person
let {name, age} = person;
console.log(name, age);

let {name: n, age:a} = person;
console.log(n, a);

//Extract an attribute of an attribute
let {address: {city, district}} = person;
console.log(city, district);

//Extract an undefined attribute
let {lastName} = person;
console.log(lastName);

//Extract and set a value to an undefined attribute
let {cpf = "111.111.111-11"} = person;
console.log(cpf);
```

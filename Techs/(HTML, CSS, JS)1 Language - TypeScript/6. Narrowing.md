# Narrowing

Narrowing is how TypeScript discovers a more specific type inside a block, from the checks in your code.

## typeof

*Example:*

```typescript
function format(value: string | number): string {
  if (typeof value === "string") {
    return value.toUpperCase();
  }
  return value.toFixed(2);
}
```

## Truthiness

*Example:*

```typescript
function printName(name?: string): void {
  if (name) {
    console.log(name.trim());
  }
}
```

## instanceof

*Example:*

```typescript
function getMessage(error: unknown): string {
  if (error instanceof Error) {
    return error.message;
  }
  return "Unknown error";
}
```

## in

*Example:*

```typescript
type Fish = { swim: () => void };
type Bird = { fly: () => void };

function move(animal: Fish | Bird): void {
  if ("swim" in animal) {
    animal.swim();
  } else {
    animal.fly();
  }
}
```

## Discriminated unions

Each type of the union has a common literal property, which identifies it.

*Example:*

```typescript
type Result =
  | { status: "success"; data: string }
  | { status: "error"; message: string };

function show(result: Result): string {
  switch (result.status) {
    case "success":
      return result.data;
    case "error":
      return result.message;
  }
}
```

## Type guards

A function that tells the compiler the type of a value, with `value is Type`.

*Example:*

```typescript
type User = { name: string };

function isUser(value: unknown): value is User {
  return typeof value === "object" && value !== null && "name" in value;
}
```

## any vs unknown vs never

- `any`: Turns off the type checking. Avoid it.
- `unknown`: Any value, but it must be narrowed before use. Prefer it for external data.
- `never`: A value that can't exist.

*Example:*

```typescript
const data: unknown = JSON.parse('{"name":"John"}');

// data.name; // Error: data is unknown

if (typeof data === "object" && data !== null && "name" in data) {
  console.log(data.name);
}
```

**Exhaustive check:** `never` makes the compiler warn when a case is forgotten.

```typescript
type Shape = "circle" | "square";

function getSides(shape: Shape): number {
  switch (shape) {
    case "circle":
      return 0;
    case "square":
      return 4;
    default: {
      const unreachable: never = shape;
      return unreachable;
    }
  }
}
```

## as const and satisfies

- `as const`: Keeps the literal values and makes everything readonly.
- `satisfies`: Checks a value against a type, without losing the specific type.

*Example:*

```typescript
const ROLES = ["admin", "user"] as const;
type Role = (typeof ROLES)[number]; // "admin" | "user"

type Colors = Record<string, string | number[]>;

const palette = {
  red: "#ff0000",
  green: [0, 255, 0],
} satisfies Colors;

palette.green.map((n) => n * 2); // still number[]
```

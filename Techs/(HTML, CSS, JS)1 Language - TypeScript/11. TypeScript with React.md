# TypeScript with React

React components use the `.tsx` extension. Create a project with the Vite `react-ts` template.

```console
npm create vite@latest my-app -- --template react-ts
```

## Props

*Example:*

```tsx
interface ButtonProps {
  label: string;
  disabled?: boolean;
  onClick: () => void;
}

export function Button({ label, disabled = false, onClick }: ButtonProps) {
  return <button disabled={disabled} onClick={onClick}>{label}</button>;
}
```

**Children:**

```tsx
import type { ReactNode } from 'react';

function Card({ children }: { children: ReactNode }) {
  return <div className="card">{children}</div>;
}
```

**Native props:** Inherit the props of an HTML element.

```tsx
import type { ComponentProps } from 'react';

interface InputProps extends ComponentProps<'input'> {
  label: string;
}

function Input({ label, ...rest }: InputProps) {
  return (
    <label>
      {label}
      <input {...rest} />
    </label>
  );
}
```

## useState

The type is inferred from the initial value. Use a generic when the value can change type.

*Example:*

```tsx
import { useState } from 'react';

interface User {
  name: string;
}

function Profile() {
  const [count, setCount] = useState(0);
  const [user, setUser] = useState<User | null>(null);

  return <button onClick={() => setCount(count + 1)}>{user?.name}</button>;
}
```

## useRef

*Example:*

```tsx
import { useRef } from 'react';

function Search() {
  const inputRef = useRef<HTMLInputElement>(null);

  return <input ref={inputRef} onFocus={() => inputRef.current?.select()} />;
}
```

## Events

*Example:*

```tsx
import type { ChangeEvent, FormEvent } from 'react';

function Form() {
  function handleChange(e: ChangeEvent<HTMLInputElement>) {
    console.log(e.target.value);
  }

  function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
  }

  return (
    <form onSubmit={handleSubmit}>
      <input onChange={handleChange} />
    </form>
  );
}
```

## Context

Use `null` as the initial value and a custom hook that checks it.

*Example:*

```tsx
import { createContext, useContext } from 'react';

interface Theme {
  color: string;
}

const ThemeContext = createContext<Theme | null>(null);

export function useTheme(): Theme {
  const theme = useContext(ThemeContext);
  if (!theme) throw new Error('useTheme must be used inside a provider');
  return theme;
}
```
